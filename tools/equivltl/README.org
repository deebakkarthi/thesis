#+TITLE: =equivltl=

=equivltl= groups a list of Linear Temporal Logic (LTL) formulas based on
equivalence.

* Dependencies
- =spot-2.16.2= =doxygen=
- =graphviz=
  
* Building =spot=
#+BEGIN_SRC bash
  ./configure --prefix=$(realpath ../opt) --disable-python
  make -j$(nproc)
  make -j$(nproc) check
  make -j$(nproc) install
#+END_SRC
